<script>
const keys = {
  notes:'digitalStash_notes',
  tasks:'digitalStash_tasks',
  trinkets:'digitalStash_trinkets',
  wishlist:'digitalStash_wishlist'
};

let data = {};
Object.keys(keys).forEach(k=>{
  try { data[k]=JSON.parse(localStorage.getItem(keys[k])) || []; }
  catch(e) { data[k]=[]; }
});

function save(){
  Object.keys(keys).forEach(k=>{
    localStorage.setItem(keys[k],JSON.stringify(data[k]));
  });
  updateStats();
}

function updateStats(){
  document.getElementById('noteCount').textContent=data.notes.length;
  document.getElementById('taskCount').textContent=data.tasks.filter(x=>!x.done).length;
  document.getElementById('trinketCount').textContent=data.trinkets.length+data.wishlist.length;
}

document.getElementById('today').textContent=
  new Date().toLocaleDateString('en-US',{weekday:'short',month:'short',day:'numeric'});

function navigate(page){
  document.querySelectorAll('.page').forEach(p=>p.classList.add('hidden'));
  document.getElementById(page).classList.remove('hidden');
  document.querySelectorAll('.nav').forEach(n=>{
    n.classList.toggle('active',n.dataset.page===page);
  });
  if(page!=='home')renderList(page);
}

document.querySelectorAll('.nav').forEach(n=>{
  n.addEventListener('click',()=>navigate(n.dataset.page));
});

let tinOpen=false;
function openTin(){
  tinOpen=!tinOpen;
  document.getElementById('tin').classList.toggle('open',tinOpen);
  document.getElementById('tinLabel').textContent=
    tinOpen?'Welcome inside! Choose a little treasure below ♡':'Tap your tin can to explore ♡';
  if(tinOpen){
    setTimeout(()=>navigate('trinkets'),500);
  }
}

function openModal(type,id=null){
  const modal=document.getElementById('modalBg');
  const existing=id?data[type].find(x=>x.id===id):null;
  document.getElementById('itemForm').reset();
  document.getElementById('itemType').value=type;
  document.getElementById('itemId').value=id||'';
  document.getElementById('modalTitle').textContent=
    existing?'Edit your little treasure ♡':'Add something new ♡';
  document.getElementById('itemTitle').value=existing?.title||'';
  document.getElementById('itemContent').value=existing?.content||'';
  document.getElementById('itemDate').value=existing?.date||'';
  document.getElementById('itemIcon').value=existing?.icon||'💌';

  document.getElementById('taskFields').classList.toggle('hidden',type!=='tasks');
  document.getElementById('trinketFields').classList.toggle('hidden',type!=='trinkets' && type!=='wishlist');
  document.getElementById('contentLabel').textContent=
    type==='tasks'?'Details (optional)':type==='wishlist'?'Why do you want it?':'Content';
  document.getElementById('itemContent').placeholder=
    type==='tasks'?'What do you need to do?':
    type==='wishlist'?'Describe your dream item...':'Write your thoughts here...';
  modal.classList.add('show');
}

function closeModal(){
  document.getElementById('modalBg').classList.remove('show');
}

document.getElementById('modalBg').addEventListener('click',e=>{
  if(e.target.id==='modalBg')closeModal();
});

document.getElementById('itemForm').addEventListener('submit',e=>{
  e.preventDefault();
  const type=document.getElementById('itemType').value;
  const id=document.getElementById('itemId').value;
  const old=id?data[type].find(x=>x.id===id):null;
  const item={
    id:id||Date.now().toString(),
    title:document.getElementById('itemTitle').value.trim(),
    content:document.getElementById('itemContent').value.trim(),
    date:document.getElementById('itemDate').value,
    icon:document.getElementById('itemIcon').value,
    done:old?.done||false,
    created:old?.created||new Date().toISOString()
  };
  if(old) Object.assign(old,item);
  else data[type].unshift(item);
  save();
  closeModal();
  renderList(type);
});

function escapeHTML(str){
  return String(str||'').replace(/[&<>"']/g,c=>({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
  })[c]);
}

function renderList(type,query=''){
  const container=document.getElementById(type+'List');
  if(!container)return;
  const filtered=data[type].filter(x=>
    (x.title+' '+x.content).toLowerCase().includes(query.toLowerCase())
  );
  if(!filtered.length){
    container.innerHTML='<div class="empty">🌷<br><br>Nothing here yet.<br>Start collecting little things!</div>';
    return;
  }
  container.innerHTML=filtered.map(x=>`
    <div class="item ${x.done?'done':''}">
      <div class="item-content">
        <h3>${type==='trinkets'||type==='wishlist'?escapeHTML(x.icon)+' ':''}${escapeHTML(x.title)}</h3>
        <p>${escapeHTML(x.content)}</p>
        ${x.date?'<small>📅 '+escapeHTML(x.date)+'</small>':''}
      </div>
      <div class="item-actions">
        ${type==='tasks'?`<button class="icon-btn" onclick="toggleTask('${x.id}')">${x.done?'↩':'✓'}</button>`:''}
        <button class="icon-btn" onclick="openModal('${type}','${x.id}')">✎</button>
        <button class="icon-btn" onclick="deleteItem('${type}','${x.id}')">×</button>
      </div>
    </div>
  `).join('');
}

function toggleTask(id){
  const item=data.tasks.find(x=>x.id===id);
  if(item)item.done=!item.done;
  save();
  renderList('tasks');
}

function deleteItem(type,id){
  if(!confirm('Remove this from your stash? ♡'))return;
  data[type]=data[type].filter(x=>x.id!==id);
  save();
  renderList(type);
}

function openTin(){
  tinOpen=!tinOpen;
  document.getElementById('tin').classList.toggle('open',tinOpen);
  document.getElementById('tinLabel').textContent=
    tinOpen?'Opening your collection... ♡':'Tap your tin can to explore ♡';
  if(tinOpen)setTimeout(()=>navigate('trinkets'),500);
}

updateStats();
Object.keys(keys).forEach(k=>renderList(k));
